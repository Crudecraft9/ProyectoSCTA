; Imprime los valores
Sev_ACTUALIZA PROC
    ; 1. numeros en pantalla
    LEA DI,Sev_tTemp
    MOV AL,Sev_temp
    CALL UTIL_NUM2
    IMPCAP_CONCOLOR Sev_tTemp,2,11,19,0,0,0EH
    LEA DI,Sev_tHum
    MOV AL,Sev_hum
    CALL UTIL_NUM2
    IMPCAP_CONCOLOR Sev_tHum,2,13,19,0,0,0EH
    CALL Sev_PROMEDIO           ; calculos con ADD, DIV, SUB y MUL

    ; temperatura: alta >= 28 , ideal 18 a 27 , baja < 18
    CMP Sev_temp,28
    JGE Sev_T1
    CMP Sev_temp,18
    JGE Sev_T2
    CALL Sev_TBAJA
    JMP Sev_HUM
Sev_T1:
    CALL Sev_TALTA
    JMP Sev_HUM
Sev_T2:
    CALL Sev_TNORMAL

    ; humedad: alta >= 56 , ideal 45 a 55 , baja < 45
Sev_HUM:
    CMP Sev_hum,56
    JGE Sev_H1
    CMP Sev_hum,45
    JGE Sev_H2
    IMPCAP_CONCOLOR Sev_vInact,16,13,60,0,0,0AH     ; BAJA
    IMPCAP_CONCOLOR Sev_vHBaja,16,16,60,0,0,0CH
    JMP Sev_DISPLAY
Sev_H1:
    IMPCAP_CONCOLOR Sev_vDesA,16,13,60,0,0,0CH      ; ALTA
    IMPCAP_CONCOLOR Sev_vHAlta,16,16,60,0,0,0CH
    JMP Sev_DISPLAY
Sev_H2:
    IMPCAP_CONCOLOR Sev_vInact,16,13,60,0,0,0AH     ; NORMAL
    IMPCAP_CONCOLOR Sev_vNorm,16,16,60,0,0,0AH

; enviar a los display
Sev_DISPLAY:
    MOV BH,0
    MOV BL,Sev_tTemp
    SUB BL,'0'
    MOV AL,Sev_tabla[BX]
    MOV DX,2030H
    OUT DX,AL
    MOV BL,Sev_tTemp+1
    SUB BL,'0'
    MOV AL,Sev_tabla[BX]
    INC DX
    OUT DX,AL
    RET
Sev_ACTUALIZA ENDP

; Calculos del dia con las 2 lecturas (manana y tarde)
Sev_PROMEDIO PROC
    ; 01. PROMEDIO = (manana + tarde) / 2
    xor ax,ax                   ; inicializar a cero
    mov al,Sev_tMan
    add al,Sev_tTar             ; ADD: AL = 20 + 30 = 50
    mov bl,2
    div bl                      ; DIV: AL = 50 / 2 = 25
    LEA DI,Sev_tProm
    CALL UTIL_NUM2
    IMPCAP_CONCOLOR Sev_tProm,2,15,19,0,0,0EH

    ; 02. VARIACION = tarde - manana
    mov al,Sev_tTar
    sub al,Sev_tMan             ; SUB: AL = 30 - 20 = 10
    LEA DI,Sev_tVar
    CALL UTIL_NUM2
    IMPCAP_CONCOLOR Sev_tVar,2,17,19,0,0,0EH

    ; 03. LECTURAS POR SEMANA = 2 x 7
    xor ax,ax
    mov al,2
    mov cl,7
    mul cl                      ; MUL: AL = 2 x 7 = 14
    LEA DI,Sev_tLec
    CALL UTIL_NUM2
    IMPCAP_CONCOLOR Sev_tLec,2,17,60,0,0,0EH
    RET
Sev_PROMEDIO ENDP
