PROC_SEVENSEG PROC
    CALL Sev_DIBUJA
    CALL Sev_ACTUALIZA
    CALL Sev_GUARDA
    CALL TECLA
    RET
PROC_SEVENSEG ENDP

Sev_GUARDA PROC
    mov ah,2CH
    int 21h
    mov al,ch
    LEA DI,Sev_rH
    CALL UTIL_NUM2
    mov al,cl
    LEA DI,Sev_rM
    CALL UTIL_NUM2
    mov al,Sev_temp
    LEA DI,Sev_rT
    CALL UTIL_NUM2
    mov al,Sev_hum
    LEA DI,Sev_rHu
    CALL UTIL_NUM2

    LEA DX,Sev_ruta
    CALL UTIL_ABRIR_ANEXAR
    JC Sev_GERR
    MOV ID,AX
    ESCRIBIR_ARCHIVO ID,40,Sev_reg
    CERRAR_ARCHIVO ID
    IMPCAP_CONCOLOR Sev_msjGOk,35,21,22,0,0,0AH
    RET
Sev_GERR:
    IMPCAP_CONCOLOR Sev_msjGEr,35,21,22,0,0,0CH
    RET
Sev_GUARDA ENDP

Sev_ACTUALIZA PROC
    LEA DI,Sev_tTemp
    MOV AL,Sev_temp
    CALL UTIL_NUM2
    IMPCAP_CONCOLOR Sev_tTemp,2,11,19,0,0,0EH
    LEA DI,Sev_tHum
    MOV AL,Sev_hum
    CALL UTIL_NUM2
    IMPCAP_CONCOLOR Sev_tHum,2,13,19,0,0,0EH
    CALL Sev_PROMEDIO

    CMP Sev_temp,28
    JGE Sev_T1
    CMP Sev_temp,18
    JGE Sev_T2
    CALL Sev_TBAJA
    JMP Sev_HUM
Sev_T1:
    CALL Sev_TALTA
    JMP Sev_HUM
Sev_T2:
    CALL Sev_TNORMAL

Sev_HUM:
    CMP Sev_hum,56
    JGE Sev_H1
    CMP Sev_hum,45
    JGE Sev_H2
    IMPCAP_CONCOLOR Sev_vInact,16,13,60,0,0,0AH
    IMPCAP_CONCOLOR Sev_vHBaja,16,16,60,0,0,0CH
    JMP Sev_DISPLAY
Sev_H1:
    IMPCAP_CONCOLOR Sev_vDesA,16,13,60,0,0,0CH
    IMPCAP_CONCOLOR Sev_vHAlta,16,16,60,0,0,0CH
    JMP Sev_DISPLAY
Sev_H2:
    IMPCAP_CONCOLOR Sev_vInact,16,13,60,0,0,0AH
    IMPCAP_CONCOLOR Sev_vNorm,16,16,60,0,0,0AH

Sev_DISPLAY:
    MOV BH,0
    MOV BL,Sev_tTemp
    SUB BL,'0'
    MOV AL,Sev_tabla[BX]
    MOV DX,2030H
    OUT DX,AL
    MOV BL,Sev_tTemp+1
    SUB BL,'0'
    MOV AL,Sev_tabla[BX]
    INC DX
    OUT DX,AL
    RET
Sev_ACTUALIZA ENDP

Sev_PROMEDIO PROC
    xor ax,ax
    mov al,Sev_tMan
    add al,Sev_tTar
    mov bl,2
    div bl
    LEA DI,Sev_tProm
    CALL UTIL_NUM2
    IMPCAP_CONCOLOR Sev_tProm,2,15,19,0,0,0EH

    mov al,Sev_tTar
    sub al,Sev_tMan
    LEA DI,Sev_tVar
    CALL UTIL_NUM2
    IMPCAP_CONCOLOR Sev_tVar,2,17,19,0,0,0EH

    xor ax,ax
    mov al,2
    mov cl,7
    mul cl
    LEA DI,Sev_tLec
    CALL UTIL_NUM2
    IMPCAP_CONCOLOR Sev_tLec,2,17,60,0,0,0EH
    RET
Sev_PROMEDIO ENDP

Sev_MOTOR PROC
    mov cx,12
    mov si,0
    mov dx,7
Sev_LISTO:
    in al,dx
    and al,10000000B
    jz Sev_LISTO
    mov al,Sev_pasos[si]
    out dx,al
    inc si
    loop Sev_LISTO
    RET
Sev_MOTOR ENDP
