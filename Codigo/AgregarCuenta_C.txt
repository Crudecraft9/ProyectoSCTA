PROC_AGREGARCUENTA PROC
Agr_INICIO:
    CALL Agr_DIBUJA
    CALL Agr_CAPTURA
    CMP Agr_esc,1               
    JE Agr_SALIR
    CMP Agr_nombre[1],0         
    JE Agr_VACIA
    CMP Agr_clave[1],0          
    JE Agr_VACIA
    MOV AL,Agr_clave[1]         
    CMP AL,Agr_conf[1]
    JE Agr_COMPARA
    JMP Agr_NOIGUAL
Agr_COMPARA:
    MOV CL,AL                   
    MOV CH,0
    LEA SI,Agr_clave+2
    LEA DI,Agr_conf+2
Agr_CICLO:
    MOV AL,[SI]
    CMP AL,[DI]
    JE Agr_IGUAL
    JMP Agr_NOIGUAL
Agr_IGUAL:
    INC SI
    INC DI
    LOOP Agr_CICLO
    JMP Agr_GUARDAR
Agr_NOIGUAL:
    IMPCAP_CONCOLOR Agr_msjClave,79,21,0,0,0,0CH
    JMP Agr_ERROR
Agr_VACIA:
    IMPCAP_CONCOLOR Agr_msjVacio,79,21,0,0,0,0CH
Agr_ERROR:
    CALL TECLA
    JMP Agr_INICIO
Agr_SALIR:
    RET

Agr_GUARDAR:
    LEA DX,Agr_ruta
    CALL UTIL_ABRIR_ANEXAR
    JC Agr_ERRARCH
    JMP Agr_ABIERTO
Agr_ERRARCH:
    IMPCAP_CONCOLOR Agr_msjErr,79,21,0,0,0,0CH
    CALL TECLA
    RET
Agr_ABIERTO:
    MOV Agr_id,AX
    ESCRIBIR_ARCHIVO Agr_id,8,Agr_etNom
    MOV BX,Agr_id
    LEA DX,Agr_nombre
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO Agr_id,2,FDat_saltoL
    ESCRIBIR_ARCHIVO Agr_id,11,Agr_etApe
    MOV BX,Agr_id
    LEA DX,Agr_apellidos
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO Agr_id,2,FDat_saltoL
    ESCRIBIR_ARCHIVO Agr_id,4,Agr_etId
    MOV BX,Agr_id
    LEA DX,Agr_ident
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO Agr_id,2,FDat_saltoL
    ESCRIBIR_ARCHIVO Agr_id,8,Agr_etCor
    MOV BX,Agr_id
    LEA DX,Agr_correo
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO Agr_id,2,FDat_saltoL
    ESCRIBIR_ARCHIVO Agr_id,10,Agr_etTel
    MOV BX,Agr_id
    LEA DX,Agr_telefono
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO Agr_id,2,FDat_saltoL
    ESCRIBIR_ARCHIVO Agr_id,7,FDat_etTur
    MOV BX,Agr_id
    LEA DX,Agr_turno
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO Agr_id,2,FDat_saltoL
    ESCRIBIR_ARCHIVO Agr_id,9,Agr_etUsu
    MOV BX,Agr_id
    LEA DX,Agr_usuario
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO Agr_id,2,FDat_saltoL
    ESCRIBIR_ARCHIVO Agr_id,12,Agr_etCla
    MOV BX,Agr_id
    LEA DX,Agr_clave
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO Agr_id,2,FDat_saltoL
    ESCRIBIR_ARCHIVO Agr_id,5,FDat_etRol
    MOV BX,Agr_id
    LEA DX,Agr_rol
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO Agr_id,2,FDat_saltoL
    ESCRIBIR_ARCHIVO Agr_id,8,Agr_etEst
    MOV BX,Agr_id
    LEA DX,Agr_estado
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO Agr_id,2,FDat_saltoL
    ESCRIBIR_ARCHIVO Agr_id,6,FDat_etZon
    MOV BX,Agr_id
    LEA DX,Agr_zona
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO Agr_id,2,FDat_saltoL
    ESCRIBIR_ARCHIVO Agr_id,22,FDat_sepa
    CERRAR_ARCHIVO Agr_id

    CMP Agr_estado[2],'1'       
    JE Agr_ACTIVA
    JMP Agr_FIN
Agr_ACTIVA:
    CALL Agr_OPERADOR
Agr_FIN:
    IMPCAP_CONCOLOR Agr_msjOk,79,21,0,0,0,0AH
    CALL TECLA
    RET
PROC_AGREGARCUENTA ENDP

Agr_DIBUJA PROC
    CALL CLEAN_SCREEN
    IMPCAP_CONCOLOR Agr_ln00,79,0,0,0,0,2FH
    IMPCAP_CONCOLOR Agr_ln02,79,2,0,0,0,0AH
    IMPCAP_CONCOLOR Agr_ln03,79,3,0,0,0,0AH
    IMPCAP_CONCOLOR Agr_ln04,79,4,0,0,0,0AH
    IMPCAP_CONCOLOR Agr_ln05,79,5,0,0,0,0AH
    IMPCAP_CONCOLOR Agr_ln06,79,6,0,0,0,0AH
    IMPCAP_CONCOLOR Agr_ln07,79,7,0,0,0,0AH
    IMPCAP_CONCOLOR Agr_ln09,79,9,0,0,0,0EH
    IMPCAP_CONCOLOR Agr_ln11,79,11,0,0,0,0EH
    IMPCAP_CONCOLOR Agr_ln12,79,12,0,0,0,0AH
    IMPCAP_CONCOLOR Agr_ln13,79,13,0,0,0,0AH
    IMPCAP_CONCOLOR Agr_ln14,79,14,0,0,0,0AH
    IMPCAP_CONCOLOR Agr_ln15,79,15,0,0,0,0AH
    IMPCAP_CONCOLOR Agr_ln16,79,16,0,0,0,0AH
    IMPCAP_CONCOLOR Agr_ln17,79,17,0,0,0,0AH
    IMPCAP_CONCOLOR Agr_ln18,79,18,0,0,0,0BH
    IMPCAP_CONCOLOR Agr_ln19,79,19,0,0,0,0AH
    IMPCAP_CONCOLOR Agr_ln21,79,21,0,0,0,0EH
    IMPCAP_CONCOLOR msjPie,80,24,0,0,0,2FH
    RET
Agr_DIBUJA ENDP

Agr_CAPTURA PROC
    MOV Agr_esc,0
    CURSOR 12,11,0
    LEA DI,Agr_nombre
    CALL Agr_LEE
    CURSOR 13,14,0
    LEA DI,Agr_apellidos
    CALL Agr_LEE
    CURSOR 14,7,0
    LEA DI,Agr_ident
    CALL Agr_LEE
    CURSOR 15,11,0
    LEA DI,Agr_correo
    CALL Agr_LEE
    CURSOR 16,13,0
    LEA DI,Agr_telefono
    CALL Agr_LEE
    CURSOR 17,29,0
    LEA DI,Agr_turno
    CALL Agr_LEE
    CURSOR 12,51,0
    LEA DI,Agr_usuario
    CALL Agr_LEE
    CURSOR 13,54,0
    LEA DI,Agr_clave
    CALL Agr_LEE
    CURSOR 14,64,0
    LEA DI,Agr_conf
    CALL Agr_LEE
    CURSOR 15,67,0
    LEA DI,Agr_rol
    CALL Agr_LEE
    CURSOR 16,69,0
    LEA DI,Agr_estado
    CALL Agr_LEE
    CURSOR 17,68,0
    LEA DI,Agr_zona
    CALL Agr_LEE
    RET
Agr_CAPTURA ENDP

Agr_LEE PROC
    CMP Agr_esc,1
    JE Agr_LEEESC
    MOV BL,0                    
    MOV SI,DI
    INC SI
    INC SI                      
    MOV CL,[DI]
    DEC CL                      
Agr_LEE1:
    CALL TECLA                  
    CMP AL,27
    JE Agr_LEEESC
    CMP AL,13
    JE Agr_LEEFIN
    CMP AL,8
    JE Agr_LEEBORRA
    CMP AL,32
    JGE Agr_LEECAR
    JMP Agr_LEE1
Agr_LEECAR:
    CMP BL,CL                   
    JGE Agr_LEE1
    MOV [SI],AL
    INC SI
    INC BL
    MOV DL,AL                   
    MOV AH,2
    INT 21H
    JMP Agr_LEE1
Agr_LEEBORRA:
    CMP BL,0
    JE Agr_LEE1
    DEC SI
    DEC BL
    MOV DL,8                    
    MOV AH,2
    INT 21H
    MOV DL,32
    MOV AH,2
    INT 21H
    MOV DL,8
    MOV AH,2
    INT 21H
    JMP Agr_LEE1
Agr_LEEFIN:
    INC DI
    MOV [DI],BL                 
    RET
Agr_LEEESC:
    MOV Agr_esc,1
    RET
Agr_LEE ENDP

Agr_OPERADOR PROC
    LEA DX,FDat_rutaOper
    CALL UTIL_ABRIR_ANEXAR
    JC Agr_OPSALE
    JMP Agr_OP1
Agr_OPSALE:
    RET
Agr_OP1:
    MOV Agr_id,AX
    ESCRIBIR_ARCHIVO Agr_id,8,FDat_etNom
    MOV BX,Agr_id
    LEA DX,Agr_usuario
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO Agr_id,2,FDat_saltoL
    ESCRIBIR_ARCHIVO Agr_id,13,FDat_etCtl
    MOV BX,Agr_id
    LEA DX,Agr_clave
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO Agr_id,2,FDat_saltoL
    ESCRIBIR_ARCHIVO Agr_id,6,FDat_etZon
    MOV BX,Agr_id
    LEA DX,Agr_zona
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO Agr_id,2,FDat_saltoL
    ESCRIBIR_ARCHIVO Agr_id,7,FDat_etTur
    MOV BX,Agr_id
    LEA DX,Agr_turno
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO Agr_id,2,FDat_saltoL
    ESCRIBIR_ARCHIVO Agr_id,5,FDat_etRol
    MOV BX,Agr_id
    LEA DX,Agr_rol
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO Agr_id,2,FDat_saltoL
    ESCRIBIR_ARCHIVO Agr_id,22,FDat_sepa
    CERRAR_ARCHIVO Agr_id
    RET
Agr_OPERADOR ENDP
