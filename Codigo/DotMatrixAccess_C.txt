Dot_ACTUALIZA PROC
    LEA DI,Dot_tTemp
    MOV AL,Dot_temp
    CALL UTIL_NUM2
    IMPCAP_CONCOLOR Dot_tTemp,6,11,57,0,0,0EH
    IMPCAP_CONCOLOR Dot_tUmb,6,12,57,0,0,0EH

    MOV AL,Dot_temp
    CMP AL,Dot_umbral
    JGE Dot_ALERTA
    LEA SI,Dot_letraN
    CALL Dot_ENVIA
    IMPCAP_CONCOLOR Dot_resN,10,19,60,0,0,0AH
    mov dx,4
    mov ax,0000100100100100B
    out dx,ax
    RET
Dot_ALERTA:
    LEA SI,Dot_letraA
    CALL Dot_ENVIA
    IMPCAP_CONCOLOR Dot_resA,10,19,60,0,0,0CH
    mov dx,4
    mov ax,0000001001001001B
    out dx,ax
    RET
Dot_ACTUALIZA ENDP

Dot_ENVIA PROC
    MOV DX,2000H
    MOV CX,5
Dot_COL:
    MOV AL,[SI]
    OUT DX,AL
    INC SI
    INC DX
    LOOP Dot_COL
    RET
Dot_ENVIA ENDP
