LOG_VALIDAR PROC
    LEA DX,FDat_rutaOper
    CALL UTIL_CARGAR
    JNC LV_LEE
    MOV Log_err,1
    STC
    RET
LV_LEE:
    CMP AX,0
    JNE LV_INI
    MOV Log_err,1
    STC
    RET
LV_INI:
    LEA SI,Util_buf
LV_SIG:
    LEA DI,FDat_etNom
    MOV CX,8
    CALL UTIL_BUSCAR
    JC LV_NO
    LEA DI,Log_bufUsr
    MOV CL,Log_lenUsr
    XOR CH,CH
    CALL UTIL_COMPARA
    JC LV_SIG
    LEA DI,FDat_etCtl
    MOV CX,13
    CALL UTIL_BUSCAR
    JC LV_NO
    LEA DI,Log_bufPas
    MOV CL,Log_lenPas
    XOR CH,CH
    CALL UTIL_COMPARA
    JC LV_SIG
    MOV BX,SI
    LEA DI,FDat_etZon
    MOV CX,6
    CALL UTIL_BUSCAR
    JC LV_ROL
    MOV AL,[SI]
    MOV Ses_zona,AL
LV_ROL:
    MOV SI,BX
    LEA DI,FDat_etRol
    MOV CX,5
    CALL UTIL_BUSCAR
    JC LV_FIN
    MOV AL,[SI]
    MOV Ses_rol,AL
LV_FIN:
    CLC
    RET
LV_NO:
    MOV Log_err,2
    STC
    RET
LOG_VALIDAR ENDP
