; Codigo pantalla Formulario
; Abre sin borrar (anexa), guarda solo los caracteres capturados,
; y usa saltos largos seguros (JNE + JMP) para evitar "jump out of range".

PROC_FORMDATOS PROC
    MOV FDat_idOper,0
    MOV FDat_idConf,0
    LEA DX,FDat_rutaOper
    CALL UTIL_ABRIR_ANEXAR
    JNC FDat_OK1
    JMP FDat_ERROR
FDat_OK1:
    MOV FDat_idOper,AX
    LEA DX,FDat_rutaConf
    CALL UTIL_ABRIR_ANEXAR
    JNC FDat_OK2
    JMP FDat_ERROR
FDat_OK2:
    MOV FDat_idConf,AX

FDat_PANTALLA:
    CALL CLEAN_SCREEN
    IMPCAP_CONCOLOR FDat_logo1,11,1,1,0,0,0AH
    IMPCAP_CONCOLOR FDat_logo2,11,2,1,0,0,0AH
    IMPCAP_CONCOLOR FDat_logo3,11,3,1,0,0,0AH
    IMPCAP_CONCOLOR FDat_logo4,11,4,1,0,0,0AH
    IMPCAP_CONCOLOR FDat_logo5,11,5,1,0,0,0AH

    ; 3. Imprimir titulo
    IMPCAP_CONCOLOR FDat_titulo1,64,1,13,0,0,0AH
    IMPCAP_CONCOLOR FDat_titulo2,64,2,13,0,0,0AH
    IMPCAP_CONCOLOR FDat_titulo3,64,3,13,0,0,0AH
    IMPCAP_CONCOLOR FDat_titulo4,64,4,13,0,0,0AH
    IMPCAP_CONCOLOR FDat_titulo5,64,5,13,0,0,0AH

    ; 1. Instrucciones
    IMPCAP_CONCOLOR FDat_msjInst,50,7,15,0,0,0EH

    ; 2. Info Personal
    IMPCAP_CONCOLOR FDat_msjDP1,38,9,1,0,0,0EH
    IMPCAP_CONCOLOR FDat_msjDP2,38,10,1,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjDP3,38,11,1,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjDP4,38,12,1,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjDP5,38,13,1,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjDP6,38,14,1,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjDP7,38,15,1,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjDP8,38,16,1,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjDP9,38,17,1,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjDP10,38,18,1,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjDP11,38,19,1,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjDP12,38,20,1,0,0,0BH
    IMPCAP_CONCOLOR FDat_msjDP13,38,21,1,0,0,0AH

    ; 3. Config Sistema
    IMPCAP_CONCOLOR FDat_msjCF1,38,9,41,0,0,0EH
    IMPCAP_CONCOLOR FDat_msjCF2,38,10,41,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjCF3,38,11,41,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjCF4,38,12,41,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjCF5,38,13,41,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjCF6,38,14,41,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjCF7,38,15,41,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjCF8,38,16,41,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjCF9,38,17,41,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjCF10,38,18,41,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjCF11,38,19,41,0,0,0AH
    IMPCAP_CONCOLOR FDat_msjCF12,38,20,41,0,0,0BH
    IMPCAP_CONCOLOR FDat_msjCF13,38,21,41,0,0,0AH

    ; 4. Pie de pagina
    IMPCAP_CONCOLOR FDat_msjPie,80,24,0,0,0,2FH

FDat_CAPTURA:
    CALL PROC_FDat_LIMPIAR              ;Limpiar buffers 

    ;Datos Operador
    CURSOR 10,11,0
    LECTURA_CADENA FDat_nombre
    CURSOR 12,16,0
    LECTURA_CADENA FDat_control
    CURSOR 14,29,0
    LECTURA_CADENA FDat_zona
    CURSOR 16,29,0
    LECTURA_CADENA FDat_turno
    CURSOR 18,28,0
    LECTURA_CADENA FDat_rol

    ;Datos configuracion
    CURSOR 10,61,0
    LECTURA_CADENA FDat_tempMax
    CURSOR 12,63,0
    LECTURA_CADENA FDat_humMax
    CURSOR 14,70,0
    LECTURA_CADENA FDat_venti
    CURSOR 16,71,0
    LECTURA_CADENA FDat_goteras

FDat_GUARDAR:
    ; operadores.txt
    ESCRIBIR_ARCHIVO FDat_idOper,8,FDat_etNom
    MOV BX,FDat_idOper
    LEA DX,FDat_nombre
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO FDat_idOper,2,FDat_saltoL
    ESCRIBIR_ARCHIVO FDat_idOper,13,FDat_etCtl
    MOV BX,FDat_idOper
    LEA DX,FDat_control
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO FDat_idOper,2,FDat_saltoL
    ESCRIBIR_ARCHIVO FDat_idOper,6,FDat_etZon
    MOV BX,FDat_idOper
    LEA DX,FDat_zona
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO FDat_idOper,2,FDat_saltoL
    ESCRIBIR_ARCHIVO FDat_idOper,7,FDat_etTur
    MOV BX,FDat_idOper
    LEA DX,FDat_turno
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO FDat_idOper,2,FDat_saltoL
    ESCRIBIR_ARCHIVO FDat_idOper,5,FDat_etRol
    MOV BX,FDat_idOper
    LEA DX,FDat_rol
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO FDat_idOper,2,FDat_saltoL
    ESCRIBIR_ARCHIVO FDat_idOper,22,FDat_sepa

    ; configuracion.txt
    ESCRIBIR_ARCHIVO FDat_idConf,18,FDat_etTmp
    MOV BX,FDat_idConf
    LEA DX,FDat_tempMax
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO FDat_idConf,2,FDat_saltoL
    ESCRIBIR_ARCHIVO FDat_idConf,20,FDat_etHum
    MOV BX,FDat_idConf
    LEA DX,FDat_humMax
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO FDat_idConf,2,FDat_saltoL
    ESCRIBIR_ARCHIVO FDat_idConf,13,FDat_etVen
    MOV BX,FDat_idConf
    LEA DX,FDat_venti
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO FDat_idConf,2,FDat_saltoL
    ESCRIBIR_ARCHIVO FDat_idConf,16,FDat_etGot
    MOV BX,FDat_idConf
    LEA DX,FDat_goteras
    CALL UTIL_ESC_BUF
    ESCRIBIR_ARCHIVO FDat_idConf,2,FDat_saltoL
    ESCRIBIR_ARCHIVO FDat_idConf,22,FDat_sepa

    IMPCAP_CONCOLOR FDat_msjOtro,55,22,14,0,0,0EH

FDat_OTRO:
    CALL TECLA
    CMP AL,'1'
    JNE FDat_O2
    JMP FDat_PANTALLA
FDat_O2:
    CMP AL,'2'
    JNE FDat_OTRO
    JMP FDat_FIN

FDat_ERROR:
    CALL PROC_FDat_CIERRA
    IMPCAP_CONCOLOR FDat_msjErr,55,22,14,0,0,0CH
    CALL TECLA
    RET

FDat_FIN:
    CALL PROC_FDat_CIERRA
    RET
PROC_FORMDATOS ENDP

; Cierra los handles que sigan abiertos
PROC_FDat_CIERRA PROC
    CMP FDat_idOper,0
    JE FDat_C2
    MOV BX,FDat_idOper
    MOV AH,3EH
    INT 21H
    MOV FDat_idOper,0
FDat_C2:
    CMP FDat_idConf,0
    JE FDat_C3
    MOV BX,FDat_idConf
    MOV AH,3EH
    INT 21H
    MOV FDat_idConf,0
FDat_C3:
    RET
PROC_FDat_CIERRA ENDP


; Limpiar los buffers
PROC_FDat_LIMPIAR PROC
    MOV CX,26
    MOV SI,2
FDat_limNom:
    MOV FDat_nombre[SI],' '
    INC SI
    LOOP FDat_limNom

    MOV CX,21
    MOV SI,2
FDat_limCtl:
    MOV FDat_control[SI],' '
    INC SI
    LOOP FDat_limCtl

    MOV CX,2
    MOV SI,2
FDat_limZon:
    MOV FDat_zona[SI],' '
    INC SI
    LOOP FDat_limZon

    MOV CX,2
    MOV SI,2
FDat_limTur:
    MOV FDat_turno[SI],' '
    INC SI
    LOOP FDat_limTur

    MOV CX,2
    MOV SI,2
FDat_limRol:
    MOV FDat_rol[SI],' '
    INC SI
    LOOP FDat_limRol

    MOV CX,16
    MOV SI,2
FDat_limTmp:
    MOV FDat_tempMax[SI],' '
    INC SI
    LOOP FDat_limTmp

    MOV CX,14
    MOV SI,2
FDat_limHum:
    MOV FDat_humMax[SI],' '
    INC SI
    LOOP FDat_limHum

    MOV CX,2
    MOV SI,2
FDat_limVen:
    MOV FDat_venti[SI],' '
    INC SI
    LOOP FDat_limVen

    MOV CX,2
    MOV SI,2
FDat_limGot:
    MOV FDat_goteras[SI],' '
    INC SI
    LOOP FDat_limGot
    RET
PROC_FDat_LIMPIAR ENDP
